## 三、AI Agent核心设计

### 3.1 Agent架构

```python
"""
AI Agent 四层架构:
1. 感知层 (Perception): 多模态输入理解
2. 理解层 (Understanding): 语义和意图理解  
3. 推理层 (Reasoning): 策略分析和决策
4. 生成层 (Generation): 输出可执行方案
"""

class VideoAnalysisAgent:
    """
    视频分析AI Agent - 主控制器
    """
    def __init__(self):
        # 初始化各层组件
        self.perception = PerceptionLayer()
        self.understanding = UnderstandingLayer()
        self.reasoning = ReasoningLayer()
        self.generation = GenerationLayer()
        
        # 知识库
        self.knowledge_base = MarketingKnowledgeBase()
        
        # Agent状态
        self.state = AgentState()
    
    async def analyze_and_adapt(
        self, 
        video_path: str,
        new_product_info: dict = None
    ) -> dict:
        """
        完整的分析和改编流程
        """
        # Phase 1: 多模态感知
        self.state.update("正在感知视频内容...")
        perception_result = await self.perception.process(video_path)
        
        # Phase 2: 深度理解
        self.state.update("正在理解营销策略...")
        understanding_result = await self.understanding.analyze(
            perception_result,
            knowledge_base=self.knowledge_base
        )
        
        # Phase 3: 策略推理
        self.state.update("正在推理最佳策略...")
        reasoning_result = await self.reasoning.think(
            perception_result,
            understanding_result,
            knowledge_base=self.knowledge_base
        )
        
        # Phase 4: 方案生成
        if new_product_info:
            self.state.update("正在生成改编方案...")
            generation_result = await self.generation.create(
                reasoning_result,
                new_product_info
            )
        else:
            generation_result = None
        
        return {
            'perception': perception_result,
            'understanding': understanding_result,
            'reasoning': reasoning_result,
            'generation': generation_result
        }
```

**职责**：将视频转化为机器可理解的多模态数据

```python
class PerceptionLayer:
    """
    感知层 - 多模态输入处理
    """
    def __init__(self):
        # 视觉处理器
        self.scene_detector = SceneDetector()
        self.image_analyzer = ImageAnalyzer()
        self.ocr_engine = OCREngine()
        
        # 音频处理器
        self.audio_extractor = AudioExtractor()
        self.speech_recognizer = SpeechRecognizer()
    
    async def process(self, video_path: str) -> dict:
        """
        多模态并行处理
        """
        # 并行执行视觉和音频处理
        visual_task = self._process_visual(video_path)
        audio_task = self._process_audio(video_path)
        
        visual_result, audio_result = await asyncio.gather(
            visual_task, 
            audio_task
        )
        
        return {
            'visual': visual_result,
            'audio': audio_result,
            'metadata': self._extract_metadata(video_path)
        }
    
    async def _process_visual(self, video_path: str) -> dict:
        """
        视觉处理流程
        """
        # 1. 镜头检测
        shots = self.scene_detector.detect(video_path)
        
        # 2. 逐镜头分析
        shot_analyses = []
        for shot in shots:
            # 提取关键帧
            keyframe = shot.get_middle_frame()
            
            # 图像分析
            analysis = {
                'objects': self.image_analyzer.detect_objects(keyframe),
                'scene': self.image_analyzer.classify_scene(keyframe),
                'text': self.ocr_engine.extract(keyframe),
                'composition': self.image_analyzer.analyze_composition(keyframe)
            }
            
            shot_analyses.append({
                'shot_info': shot.to_dict(),
                'analysis': analysis
            })
        
        return {
            'shots': shot_analyses,
            'total_shots': len(shots),
            'video_duration': shots[-1].end_time if shots else 0
        }
    
    async def _process_audio(self, video_path: str) -> dict:
        """
        音频处理流程
        """
        # 1. 提取音频
        audio_path = self.audio_extractor.extract(video_path)
        
        # 2. 语音识别
        transcript = self.speech_recognizer.transcribe(
            audio_path,
            language='zh',
            return_timestamps=True
        )
        
        return {
            'transcript': transcript['text'],
            'segments': transcript['segments'],
            'language': transcript['language']
        }
```

```python
class SceneDetector:
    """镜头切换检测"""
    def __init__(self):
        from scenedetect import detect, ContentDetector
        self.detector = ContentDetector(threshold=27.0)
    
    def detect(self, video_path: str) -> List[Shot]:
        """检测镜头切换点"""
        from scenedetect import detect
        scene_list = detect(video_path, self.detector)
        
        shots = []
        for i, (start, end) in enumerate(scene_list):
            shot = Shot(
                index=i,
                start_time=start.get_seconds(),
                end_time=end.get_seconds(),
                start_frame=start.get_frames(),
                end_frame=end.get_frames()
            )
            shots.append(shot)
        
        return shots

class ImageAnalyzer:
    """图像分析器 - 基于CLIP"""
    def __init__(self):
        from transformers import CLIPProcessor, CLIPModel
        self.model = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
        self.processor = CLIPProcessor.from_pretrained("openai/clip-vit-base-patch32")
        
        # 预定义的分类标签
        self.shot_type_labels = [
            "person talking to camera",
            "product display",
            "close-up detail",
            "hands using product",
            "call to action"
        ]
    
    def classify_scene(self, image: np.ndarray) -> dict:
        """场景分类 (零样本)"""
        from PIL import Image
        
        pil_image = Image.fromarray(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
        
        inputs = self.processor(
            text=self.shot_type_labels,
            images=pil_image,
            return_tensors="pt",
            padding=True
        )
        
        with torch.no_grad():
            outputs = self.model(**inputs)
            probs = outputs.logits_per_image.softmax(dim=1)[0]
        
        # 获取最高概率的类别
        max_idx = probs.argmax().item()
        
        return {
            'type': self.shot_type_labels[max_idx],
            'confidence': float(probs[max_idx]),
            'all_scores': {
                label: float(score) 
                for label, score in zip(self.shot_type_labels, probs)
            }
        }
    
    def detect_objects(self, image: np.ndarray) -> List[dict]:
        """目标检测 - 使用YOLOv8"""
        from ultralytics import YOLO
        
        model = YOLO('yolov8n.pt')
        results = model(image)
        
        detections = []
        for r in results:
            for box in r.boxes:
                detections.append({
                    'class': r.names[int(box.cls[0])],
                    'confidence': float(box.conf[0]),
                    'bbox': box.xyxy[0].tolist()
                })
        
        return detections

class ReasoningLayer:

            'replication_score': replication_score,

        }
    
    async def _extract_pattern(
        self,
        perception_result: dict,
        understanding_result: dict
    ) -> dict:
        """
        提取可复用的结构模式
        """
        prompt = f"""
你是视频营销策略大师。基于对这个爆款视频的深度分析，请提取出可复用的结构模式。

分析结果:
{json.dumps(understanding_result, ensure_ascii=False, indent=2)}

请提取:
1. 脚本框架模板 (抽象化的结构)
2. 每个阶段的关键要素
3. 文案创作公式
4. 适用的产品类型和场景

返回一个结构化的模板，要求:
- 高度抽象，可应用到不同产品
- 保留核心的成功逻辑
- 提供明确的创作指南

以JSON格式返回模板。
"""
        
        response = await self.llm.chat.completions.create(
            messages=[
                {"role": "system", "content": "你是营销策略专家"},

            temperature=0.3  # 降低随机性，提高一致性

    async def _generate_recommendations(
        self,
        perception_result: dict,
        understanding_result: dict,
        knowledge_base: MarketingKnowledgeBase

        生成优化建议

            perception_result,
            knowledge_base

        # LLM生成深度建议

            perception_result,
            understanding_result

        # 合并去重
        all_suggestions = rule_based_suggestions + llm_suggestions
        
        return self._rank_suggestions(all_suggestions)
    
    def _apply_rules(

                'expected_impact': "转化率可能提升10-15%",
                'source': 'best_practice_rule'
            })
        
        # 规则3: 镜头节奏

class GenerationLayer:
    """
    生成层 - 改编方案生成
    """
    def __init__(self):
        self.llm = OpenAI(model="gpt-4-turbo-preview")
        self.prompt_builder = AdaptationPromptBuilder()
    
    async def create(
        self,
        reasoning_result: dict,
        new_product_info: dict
    ) -> dict:
        """
        生成改编脚本
        """
        # 1. 逐镜头改编
        adapted_shots = []
        
        pattern = reasoning_result['pattern']
        
        for shot_template in pattern['shot_templates']:
            adapted_shot = await self._adapt_single_shot(
                shot_template,
                new_product_info,
                pattern['guidelines']
            )
            adapted_shots.append(adapted_shot)
        
        # 2. 整体优化
        optimized_script = await self._optimize_script(
            adapted_shots,
            new_product_info
        )
        
        # 3. 生成执行文档
        execution_docs = self._generate_execution_docs(
            optimized_script,
            new_product_info
        )
        
        return {
            'adapted_script': optimized_script,
            'execution_docs': execution_docs,
            'metadata': {
                'estimated_duration': self._calculate_duration(optimized_script),
                'difficulty_score': self._assess_difficulty(optimized_script),
                'budget_estimate': self._estimate_budget(optimized_script)
            }
        }
    
    async def _adapt_single_shot(
        self,
        shot_template: dict,
        new_product_info: dict,
        guidelines: dict
    ) -> dict:
        """
        改编单个镜头
        """
        prompt = self.prompt_builder.build_shot_adaptation_prompt(
            shot_template=shot_template,
            product_category=new_product_info['category'],
            product_name=new_product_info['name'],
            selling_points=new_product_info['selling_points'],
            brand_tone=new_product_info.get('brand_tone', 'professional'),
            guidelines=guidelines
        )
        
        response = await self.llm.chat.completions.create(
            messages=[
                {
                    "role": "system", 
                    "content": "你是专业的短视频文案编剧和创意策划"
                },
                {"role": "user", "content": prompt}
            ],
            response_format={"type": "json_object"},
            temperature=0.7
        )
        
        result = json.loads(response.choices[0].message.content)
        
        return {
            'shot_index': shot_template['index'],
            'shot_type': shot_template['type'],
            'time_range': shot_template['time_range'],
            'original_purpose': shot_template['purpose'],
            'adapted_content': {
                'scene_description_cn': result['scene_description_cn'],
                'scene_description_en': result['scene_description_en'],
                'transcript_cn': result['transcript_cn'],
                'transcript_en': result['transcript_en'],
                'text_overlay': result.get('text_overlay', ''),
                'shooting_tips': result.get('shooting_tips', [])
            },
            'confidence': result.get('adaptation_confidence', 0.8)
        }
```

class AgentState:
    """
    Agent状态管理 - 追踪推理过程
    """
    def __init__(self):
        self.current_phase = None
        self.progress = 0
        self.reasoning_trace = []
        self.intermediate_results = {}
    
    def update(self, message: str, progress: int = None):
        """更新状态"""
        self.reasoning_trace.append({
            'timestamp': datetime.now(),
            'message': message,
            'phase': self.current_phase
        })
        
        if progress:
            self.progress = progress
        
        # 实时推送给前端（WebSocket）
        self._notify_frontend()
    
    def save_intermediate(self, key: str, value: any):
        """保存中间结果"""
        self.intermediate_results[key] = value
    
    def get_trace(self) -> List[dict]:
        """获取完整的推理轨迹"""
        return self.reasoning_trace

### 8.1 缓存策略

```python
from functools import lru_cache
import hashlib

class CachedAnalysis:
    """分析结果缓存"""
    def __init__(self, redis_client):
        self.redis = redis_client
        self.ttl = 86400  # 24小时
    
    def get_cache_key(self, video_path: str) -> str:
        """生成缓存key"""
        with open(video_path, 'rb') as f:
            file_hash = hashlib.md5(f.read()).hexdigest()
        return f"analysis:{file_hash}"
    
    async def get(self, video_path: str) -> dict:
        """获取缓存"""
        key = self.get_cache_key(video_path)
        cached = await self.redis.get(key)
        
        if cached:
            return json.loads(cached)
        return None
    
    async def set(self, video_path: str, result: dict):
        """设置缓存"""
        key = self.get_cache_key(video_path)
        await self.redis.setex(
            key, 
            self.ttl, 
            json.dumps(result)
        )
```
